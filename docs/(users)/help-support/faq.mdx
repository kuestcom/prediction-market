import { Accordions, Accordion } from 'fumadocs-ui/components/accordion'

<Accordions type="single" collapsible>
  <Accordion title="What fee do I pay?" value="fees">
    Fees vary by market and depend on execution price and filled quantity. Review the
    order preview before signing and Portfolio History after a fill. A referral does not
    change the fee paid by the trader.
  </Accordion>

<Accordion title="Why did my Market order fill at more than one price?" value="market-fill">
  A Market order can consume several order-book levels. Its average execution price depends on the liquidity available
  when it reaches the CLOB.
</Accordion>

<Accordion title="Where do I cancel an order?" value="cancel-orders">
  Open <strong>Portfolio → Open Orders</strong>. You can cancel unfilled Limit quantity or use{' '}
  <strong>Cancel all</strong> when the action is available.
</Accordion>

<Accordion title="When can I claim winnings?" value="claim">
  Claim after the market reaches final resolution. Use the Portfolio winnings card or
  <strong>Claim winnings</strong> in the resolved market panel.
</Accordion>

<Accordion title="What happens while a market is disputed?" value="dispute">
  The result is not final and positions are not claimable. Monitor the market timeline through the dispute and
  final-review states.
</Accordion>

<Accordion title="How do I claim affiliate fees?" value="affiliate-claim">
  Open <strong>Settings → Affiliate</strong> and select <strong>Claim</strong> in the Affiliate Commissions card when a
  balance is available.
</Accordion>

  <Accordion title="How do I export activity?" value="csv">
    Open <strong>Portfolio → History</strong>, apply the filters you need, and select the
    download icon. The CSV reflects the current filtered view.
  </Accordion>
</Accordions>
